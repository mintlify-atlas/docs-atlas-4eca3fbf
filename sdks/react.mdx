## Setup

### 1. Add the Provider

Wrap your application with `AutumnProvider`:

```tsx App.tsx
import { AutumnProvider } from 'autumn-js/react';

function App() {
  return (
    <AutumnProvider
      backendUrl="https://api.useautumn.com/v1"
      publicKey="am_pk_test_..."
    >
      {/* Your app components */}
    </AutumnProvider>
  );
}
```

### Provider Props

<ParamField path="backendUrl" type="string" required>
  Your Autumn API URL. Use `https://api.useautumn.com/v1` for production.
</ParamField>

<ParamField path="publicKey" type="string">
  Your Autumn public key for client-side authentication. Optional if using credential-based auth.
</ParamField>

<ParamField path="includeCredentials" type="boolean" default="false">
  Set to `true` for session-based authentication.
</ParamField>

## Using the Hook

Access Autumn functions using the `useAutumn` hook:

```tsx
import { useAutumn } from 'autumn-js/react';

function MyComponent() {
  const { attach, check, track } = useAutumn();
  
  // Use attach, check, and track functions
}
```

## Core Operations

### Attach (Subscribe to Plans)

Handle product subscriptions and upgrades:

```tsx PricingCard.tsx
import { useAutumn } from 'autumn-js/react';
import { useState } from 'react';

function PricingCard({ productId, customerId }) {
  const { attach } = useAutumn();
  const [loading, setLoading] = useState(false);

  const handleSubscribe = async () => {
    setLoading(true);
    try {
      const { data } = await attach({
        customerId,
        productId,
        successUrl: window.location.origin + '/success',
        cancelUrl: window.location.origin + '/pricing'
      });
      
      // For new customers, data.url contains Stripe Checkout URL
      if (data.url) {
        window.location.href = data.url;
      }
    } catch (error) {
      console.error('Subscription failed:', error);
    } finally {
      setLoading(false);
    }
  };

  return (
    <button onClick={handleSubscribe} disabled={loading}>
      {loading ? 'Processing...' : 'Subscribe'}
    </button>
  );
}
```

### Attach with Quantity

Handle seat-based pricing:

```tsx
const { attach } = useAutumn();

await attach({

  quantity: 5 // Number of seats

```

### Check (Verify Access)

Check if a customer can access a feature:

```tsx FeatureGate.tsx
import { useAutumn } from 'autumn-js/react';
import { useEffect, useState } from 'react';

function FeatureGate({ customerId, featureId, children }) {
  const { check } = useAutumn();
  const [hasAccess, setHasAccess] = useState(false);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const checkAccess = async () => {
      try {
        const { data } = await check({ customerId, featureId });
        setHasAccess(data.allowed);
      } catch (error) {
        console.error('Access check failed:', error);
        setHasAccess(false);
      } finally {
        setLoading(false);
      }
    };

    checkAccess();
  }, [customerId, featureId]);

  if (loading) return <div>Checking access...</div>;
  if (!hasAccess) return <div>Upgrade to access this feature</div>;
  
  return <>{children}</>;
}
```

### Check with Usage Information

Display remaining usage limits:

```tsx UsageDisplay.tsx
import { useAutumn } from 'autumn-js/react';
import { useEffect, useState } from 'react';

function UsageDisplay({ customerId, featureId }) {
  const { check } = useAutumn();
  const [usage, setUsage] = useState(null);

  useEffect(() => {
    const fetchUsage = async () => {
      const { data } = await check({ 
        customerId, 
        featureId
      });
      
      setUsage({
        allowed: data.allowed,
        remaining: data.remaining,
        limit: data.limit,
        used: data.current_usage
      });
    };

    fetchUsage();
  }, [customerId, featureId]);

  if (!usage) return null;

  return (
    <div className="usage-display">
      <div className="progress-bar">
        <div 
          className="progress" 
          style={{ width: `${(usage.used / usage.limit) * 100}%` }}
        />
      </div>
      <p>Used: {usage.used} / {usage.limit}</p>
      <p>Remaining: {usage.remaining}</p>
      {!usage.allowed && (
        <p className="warning">Limit reached! Upgrade to continue.</p>
      )}
    </div>
  );
}
```

### Track (Record Usage)

Track feature consumption:

```tsx ApiCallButton.tsx
import { useAutumn } from 'autumn-js/react';
import { useState } from 'react';

function ApiCallButton({ customerId }) {
  const { check, track } = useAutumn();
  const [loading, setLoading] = useState(false);

  const handleApiCall = async () => {
    setLoading(true);
    try {
      // Check if customer has access
      const { data } = await check({
        customerId,
        featureId: 'api_calls'
      });

      if (!data.allowed) {
        alert('API limit reached. Please upgrade your plan.');
        return;
      }

      // Perform the API call
      await performApiOperation();

      // Track the usage
      await track({
        customerId,
        featureId: 'api_calls',
        value: 1
      });
    } finally {
      setLoading(false);
    }
  };

  return (
    <button onClick={handleApiCall} disabled={loading}>
      {loading ? 'Processing...' : 'Make API Call'}
    </button>
  );
}
```

## Custom Hooks

### useFeatureAccess Hook

Create a reusable hook for feature checks:

```tsx hooks/useFeatureAccess.ts
import { useAutumn } from 'autumn-js/react';
import { useEffect, useState } from 'react';

interface FeatureAccess {
  allowed: boolean;
  remaining?: number;
  used?: number;
  limit?: number;
  loading: boolean;
  error?: Error;
}

export function useFeatureAccess(
  customerId: string,
  featureId: string
): FeatureAccess {
  const { check } = useAutumn();
  const [access, setAccess] = useState<FeatureAccess>({
    allowed: false,
    loading: true
  });

  useEffect(() => {
    const checkAccess = async () => {
      try {
        const { data } = await check({
          customerId,
          featureId
        });
        
        setAccess({
          allowed: data.allowed,
          remaining: data.remaining,
          used: data.current_usage,
          limit: data.limit,
          loading: false
        });
      } catch (error) {
        setAccess({ 
          allowed: false, 
          loading: false,
          error: error as Error
        });
      }
    };

    checkAccess();
  }, [customerId, featureId, check]);

  return access;
}
```

Usage:

```tsx
import { useFeatureAccess } from './hooks/useFeatureAccess';

function MyComponent({ customerId }) {
  const { allowed, remaining, loading, error } = useFeatureAccess(
    customerId,
    'api_calls'
  );

  if (loading) return <div>Loading...</div>;
  if (error) return <div>Error loading access</div>;
  
  return (
    <div>
      {allowed ? (
        <p>You have {remaining} API calls remaining</p>
      ) : (
        <a href="/pricing">Upgrade to continue</a>
      )}
    </div>
  );
}
```

### useProducts Hook

Fetch and display products:

```tsx hooks/useProducts.ts
import { useAutumn } from 'autumn-js/react';
import { useEffect, useState } from 'react';
import type { Product } from 'autumn-js';

export function useProducts(customerId?: string) {
  const { products } = useAutumn();
  const [productList, setProductList] = useState<Product[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const fetchProducts = async () => {
      try {
        const { data } = await products.list({ customerId });
        setProductList(data.list);
      } finally {
        setLoading(false);
      }
    };

    fetchProducts();
  }, [customerId]);

  return { products: productList, loading };
}
```

Usage:

```tsx PricingPage.tsx
import { useProducts } from './hooks/useProducts';

function PricingPage({ customerId }) {
  const { products, loading } = useProducts(customerId);

  if (loading) return <div>Loading plans...</div>;

  return (
    <div className="pricing-grid">
      {products.map(product => (
        <ProductCard key={product.id} product={product} />
      ))}
    </div>
  );
}
```

## Advanced Patterns

### Checkout with Preview

Show pricing preview before confirming subscription:

```tsx CheckoutFlow.tsx
import { useAutumn } from 'autumn-js/react';
import { useState } from 'react';

function CheckoutFlow({ customerId, productId }) {
  const { checkout, attach } = useAutumn();
  const [preview, setPreview] = useState(null);
  const [loading, setLoading] = useState(false);

  const handleCheckout = async () => {
    setLoading(true);
    try {
      const { data } = await checkout({ customerId, productId });
      
      if (data.url) {
        // New customer - redirect to Stripe
        window.location.href = data.url;
      } else {
        // Returning customer - show preview
        setPreview(data);
      }
    } finally {
      setLoading(false);
    }
  };

  const handleConfirm = async () => {
    setLoading(true);
    try {
      await attach({ customerId, productId });
      // Subscription updated - refresh or redirect
      window.location.reload();
    } finally {
      setLoading(false);
    }
  };

  if (preview) {
    return (
      <div className="checkout-preview">
        <h3>Confirm Subscription Change</h3>
        <div className="preview-details">
          <p>Plan: {preview.product.name}</p>
          <p>Price: {preview.total / 100} {preview.currency.toUpperCase()}</p>
          {preview.prorated && (
            <p>Prorated adjustment: {preview.prorated / 100}</p>
          )}
        </div>
        <div className="actions">
          <button onClick={() => setPreview(null)}>Cancel</button>
          <button onClick={handleConfirm} disabled={loading}>
            {loading ? 'Confirming...' : 'Confirm'}
          </button>
        </div>
      </div>
    );
  }

  return (
    <button onClick={handleCheckout} disabled={loading}>
      {loading ? 'Processing...' : 'Subscribe'}
    </button>
  );
}
```

### Protected Component

Create a component that only renders if access is granted:

```tsx components/Protected.tsx
import { useFeatureAccess } from '../hooks/useFeatureAccess';
import type { ReactNode } from 'react';

interface ProtectedProps {
  customerId: string;
  featureId: string;
  children: ReactNode;
  fallback?: ReactNode;
}

export function Protected({ 
  customerId, 
  featureId, 
  children,
  fallback 
}: ProtectedProps) {
  const { allowed, loading } = useFeatureAccess(customerId, featureId);

  if (loading) {
    return <div className="loading-spinner" />;
  }

  if (!allowed) {
    return fallback || (
      <div className="upgrade-prompt">
        <h3>Upgrade Required</h3>
        <p>This feature requires a premium plan.</p>
        <a href="/pricing">View Plans</a>
      </div>
    );
  }

  return <>{children}</>;
}
```

Usage:

```tsx
import { Protected } from './components/Protected';

function Dashboard({ customerId }) {
  return (
    <div>
      <h1>Dashboard</h1>
      
      <Protected customerId={customerId} featureId="analytics">
        <AnalyticsPanel />
      </Protected>
      
      <Protected 
        customerId={customerId} 
        featureId="export"
        fallback={<div>Upgrade to export data</div>}
      >
        <ExportButton />
      </Protected>
    </div>
  );
}
```

### Usage Meter Component

Display real-time usage with progress bar:

```tsx components/UsageMeter.tsx
import { useFeatureAccess } from '../hooks/useFeatureAccess';

interface UsageMeterProps {
  customerId: string;
  featureId: string;
  label: string;
}

export function UsageMeter({ customerId, featureId, label }: UsageMeterProps) {
  const { used, limit, remaining, loading } = useFeatureAccess(
    customerId,
    featureId
  );

  if (loading) return <div>Loading...</div>;

  const percentage = limit ? (used / limit) * 100 : 0;
  const isNearLimit = percentage > 80;
  const isAtLimit = remaining === 0;

  return (
    <div className="usage-meter">
      <div className="header">
        <span className="label">{label}</span>
        <span className="count">
          {used.toLocaleString()} / {limit.toLocaleString()}
        </span>
      </div>
      
      <div className="progress-bar">
        <div 
          className={`progress ${isAtLimit ? 'error' : isNearLimit ? 'warning' : ''}`}
          style={{ width: `${Math.min(percentage, 100)}%` }}
        />
      </div>
      
      <div className="footer">
        <span className="remaining">
          {remaining.toLocaleString()} remaining
        </span>
        {isAtLimit && (
          <a href="/pricing" className="upgrade-link">
            Upgrade
          </a>
        )}
      </div>
    </div>
  );
}
```

## React Query Integration

Use React Query for caching and automatic refetching:

```tsx hooks/useFeatureQuery.ts
import { useQuery } from '@tanstack/react-query';
import { useAutumn } from 'autumn-js/react';

export function useFeatureQuery(customerId: string, featureId: string) {
  const { check } = useAutumn();
  
  return useQuery({
    queryKey: ['feature', customerId, featureId],
    queryFn: () => check({ customerId, featureId }),
    staleTime: 60000, // Cache for 1 minute
    retry: 2
  });
}
```

Usage:

```tsx
import { useFeatureQuery } from './hooks/useFeatureQuery';

function MyComponent({ customerId }) {
  const { data, isLoading, error } = useFeatureQuery(
    customerId,
    'api_calls'
  );

  if (isLoading) return <div>Loading...</div>;
  if (error) return <div>Error: {error.message}</div>;
  
  return (
    <div>
      {data.data.allowed ? 'Access granted' : 'Upgrade required'}
    </div>
  );
}
```

## Best Practices

<AccordionGroup>
  <Accordion title="Use Environment Variables">
    Never hardcode API keys:
    
    ```tsx
    <AutumnProvider
      publicKey={import.meta.env.VITE_AUTUMN_PUBLIC_KEY}
      backendUrl={import.meta.env.VITE_AUTUMN_API_URL}
    >
    ```
  </Accordion>

  <Accordion title="Handle Loading States">
    Always show loading states:
    
    ```tsx
    const [loading, setLoading] = useState(false);
    
    const handleAction = async () => {
      setLoading(true);
      try {
        await autumn.track(...);
      } finally {
        setLoading(false);
      }
    };
    ```
  </Accordion>

  <Accordion title="Implement Error Boundaries">
    Wrap components in error boundaries:
    
    ```tsx
    <ErrorBoundary fallback={<ErrorMessage />}>
      <Protected customerId={id} featureId="premium">
        <PremiumFeature />
      </Protected>
    </ErrorBoundary>
    ```
  </Accordion>

  <Accordion title="Cache with React Query">
    Use React Query for caching:
    
    ```tsx
    import { useQuery } from '@tanstack/react-query';
    
    function useFeatureCheck(customerId: string, featureId: string) {
      const { check } = useAutumn();
      
      return useQuery({
        queryKey: ['feature', customerId, featureId],
        queryFn: () => check({ customerId, featureId }),
        staleTime: 60000
      });
    }
    ```
  </Accordion>

  <Accordion title="Debounce Track Calls">
    Debounce frequent tracking:
    
    ```tsx
    import { useDebouncedCallback } from 'use-debounce';
    
    const debouncedTrack = useDebouncedCallback(
      (customerId, featureId, value) => {
        autumn.track({ customerId, featureId, value });
      },
      1000
    );
    ```
  </Accordion>
