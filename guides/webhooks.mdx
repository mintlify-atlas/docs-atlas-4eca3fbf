## Overview

Webhooks notify your application when important events happen in Autumn. Use webhooks to:

- **Provision access** - Grant features when customers subscribe
- **Handle cancellations** - Revoke access when subscriptions end
- **Process payments** - React to successful or failed payments
- **Sync data** - Keep your database in sync with billing events
- **Send notifications** - Email customers about billing changes
- **Update analytics** - Track subscription metrics in real-time

Autumn sends HTTP POST requests to your webhook endpoint using Svix for secure, reliable delivery.

## Setting Up Webhooks

<Steps>
  <Step title="Create a webhook endpoint">
    Create an endpoint in your application to receive webhook events:

    ```typescript
    import { Webhook } from "svix";

    app.post("/webhooks/autumn", async (req, res) => {
      const rawBody = await req.text();
      const headers = req.headers;

      // Verify the webhook signature
      const wh = new Webhook(process.env.AUTUMN_WEBHOOK_SECRET);
      
      try {
        const event = wh.verify(rawBody, {
          "svix-id": headers["svix-id"],
          "svix-timestamp": headers["svix-timestamp"],
          "svix-signature": headers["svix-signature"]
        });

        // Process the event
        await handleWebhookEvent(event);

        return res.json({ success: true });
      } catch (err) {
        return res.status(400).json({ error: "Invalid signature" });
      }
    });
    ```
  </Step>

  <Step title="Configure in dashboard">
    Add your webhook URL in the Autumn dashboard:

    1. Go to Settings → Webhooks
    2. Click "Add Endpoint"
    3. Enter your webhook URL (e.g., `https://yourapp.com/webhooks/autumn`)
    4. Select which events to receive
    5. Copy your webhook signing secret
  </Step>

  <Step title="Test the integration">
    Send a test event from the dashboard to verify your endpoint is working.
  </Step>
</Steps>

## Webhook Events

### Customer Product Events

**`customer.product.created`** - Customer subscribed to a product

```typescript
{
  type: "customer.product.created",
  data: {
    customer_id: "customer_123",
    product_id: "pro_plan",
    status: "active",
    scenario: "subscription_started",
    updated_product: {
      id: "pro_plan",
      name: "Pro Plan"
    }
  }
}
```

**`customer.product.updated`** - Product status or configuration changed

```typescript
{
  type: "customer.product.updated",
  data: {
    customer_id: "customer_123",
    product_id: "pro_plan",
    status: "active",
    scenario: "subscription_renewed",
    updated_product: {
      id: "pro_plan",
      name: "Pro Plan"
    }
  }
}
```

Common scenarios:
- `subscription_started` - New subscription activated
- `subscription_renewed` - Subscription renewed for new period
- `subscription_upgraded` - Customer upgraded to higher tier
- `subscription_downgraded` - Customer downgraded to lower tier
- `subscription_canceled` - Subscription canceled

**`customer.product.deleted`** - Subscription canceled or expired

```typescript
{
  type: "customer.product.deleted",
  data: {
    customer_id: "customer_123",
    product_id: "pro_plan",
    status: "canceled",
    canceled_at: "2026-02-17T12:00:00Z"
  }
}
```

### Usage Events

**`customer.threshold_reached`** - Customer hit usage threshold

```typescript
{
  type: "customer.threshold_reached",
  data: {
    customer_id: "customer_123",
    feature: {
      id: "api_calls",
      name: "API Calls"
    },
    threshold: 90,
    current_usage: 9000,
    limit: 10000
  }
}
```

## Event Handling Patterns

### Provisioning Access

Grant access when customers subscribe:

```typescript
async function handleWebhookEvent(event: { type: string; data: any }) {
  switch (event.type) {
    case "customer.product.created":
      if (event.data.scenario === "subscription_started") {
        await provisionAccess({
          customerId: event.data.customer_id,
          productId: event.data.product_id
        });
      }
      break;
  }
}

async function provisionAccess({ customerId, productId }: {
  customerId: string;
  productId: string;
}) {
  // Enable features in your system
  await db.users.update(
    { autumnCustomerId: customerId },
    { plan: productId, accessGranted: true }
  );

  // Get user email
  const user = await db.users.findOne({ autumnCustomerId: customerId });

  // Send welcome email
  await sendEmail({
    to: user.email,
    subject: "Welcome to Pro!",
    template: "subscription_started",
    data: { productId }
  });
}
```

### Revoking Access

Remove access when subscriptions end:

```typescript
case "customer.product.deleted":
  if (event.data.status === "canceled") {
    await revokeAccess({
      customerId: event.data.customer_id,
      productId: event.data.product_id
    });
  }
  break;

async function revokeAccess({ customerId, productId }: {
  customerId: string;
  productId: string;
}) {
  // Disable features
  await db.users.update(
    { autumnCustomerId: customerId },
    { plan: "free", accessGranted: false }
  );

  // Get user email
  const user = await db.users.findOne({ autumnCustomerId: customerId });

  // Send cancellation email
  await sendEmail({
    to: user.email,
    subject: "Subscription Canceled",
    template: "subscription_canceled"
  });
}
```

### Handling Upgrades and Downgrades

Update access when customers change plans:

```typescript
case "customer.product.updated":
  const { scenario, customer_id, updated_product } = event.data;
  
  if (scenario === "subscription_upgraded") {
    // Grant additional features
    await updateFeatureAccess(customer_id, updated_product.id);
    
    await sendEmail({
      to: await getCustomerEmail(customer_id),
      template: "plan_upgraded",
      data: { plan: updated_product.name }
    });
  }
  
  if (scenario === "subscription_downgraded") {
    // Remove premium features (may happen at period end)
    await updateFeatureAccess(customer_id, updated_product.id);
    
    await sendEmail({
      to: await getCustomerEmail(customer_id),
      template: "plan_downgraded",
      data: { plan: updated_product.name }
    });
  }
  break;
```

### Usage Alerts

Notify customers when approaching limits:

```typescript
case "customer.threshold_reached":
  const { customer_id, feature, threshold, current_usage, limit } = event.data;
  
  const customer = await autumn.customers.get(customer_id);
  
  await sendEmail({
    to: customer.email,
    subject: `${threshold}% of ${feature.name} limit reached`,
    template: "usage_threshold",
    data: {
      feature: feature.name,
      used: current_usage,
      limit: limit,
      remaining: limit - current_usage,
      upgradeUrl: `https://yourapp.com/upgrade`
    }
  });
  break;
```

## Webhook Verification

Always verify webhook signatures using Svix:

```typescript
import { Webhook } from "svix";

function verifyWebhook(
  rawBody: string,
  headers: Record<string, string | undefined>
) {
  const wh = new Webhook(process.env.AUTUMN_WEBHOOK_SECRET!);
  
  const svixId = headers["svix-id"];
  const svixTimestamp = headers["svix-timestamp"];
  const svixSignature = headers["svix-signature"];

  if (!svixId || !svixTimestamp || !svixSignature) {
    throw new Error("Missing svix headers");
  }
  
  try {
    const event = wh.verify(rawBody, {
      "svix-id": svixId,
      "svix-timestamp": svixTimestamp,
      "svix-signature": svixSignature
    });
    return event as { type: string; data: Record<string, unknown> };
  } catch (err) {
    throw new Error("Invalid webhook signature");
  }
}
```

<Warning>
Never process unverified webhooks. Attackers could send fake events to manipulate your system.
</Warning>

## Retry Logic

Autumn automatically retries failed webhook deliveries:

- Immediate retry
- After 5 minutes
- After 15 minutes  
- After 1 hour
- After 3 hours
- After 6 hours

Return a 2xx status code to acknowledge receipt and stop retries.

```typescript
app.post("/webhooks/autumn", async (req, res) => {
  try {
    const event = verifyWebhook(req.body, req.headers);
    
    // Process asynchronously
    processWebhookAsync(event).catch(console.error);
    
    // Acknowledge immediately
    return res.status(200).json({ received: true });
  } catch (err) {
    console.error("Webhook verification error:", err);
    return res.status(400).json({ error: err.message });
  }
});
```

<Tip>
Respond quickly (within 3 seconds) to acknowledge the webhook, then process events asynchronously to avoid timeouts.
</Tip>

## Idempotency

Webhooks may be delivered more than once. Make your handlers idempotent:

```typescript
async function handleWebhookEvent(event: any) {
  // Check if already processed
  const existing = await db.webhookEvents.findOne({
    eventId: event.id
  });

  if (existing) {
    console.log("Event already processed:", event.id);
    return;
  }

  // Process event
  await processEvent(event);

  // Mark as processed
  await db.webhookEvents.insert({
    eventId: event.id,
    type: event.type,
    processedAt: new Date()
  });
}
```

## Complete Example

Full webhook handler with all patterns:

```typescript
import express from "express";
import { Webhook } from "svix";

const app = express();

app.post("/webhooks/autumn", 
  express.raw({ type: "application/json" }),
  async (req, res) => {
    try {
      // Verify webhook
      const wh = new Webhook(process.env.AUTUMN_WEBHOOK_SECRET!);
      const event = wh.verify(req.body.toString(), {
        "svix-id": req.headers["svix-id"] as string,
        "svix-timestamp": req.headers["svix-timestamp"] as string,
        "svix-signature": req.headers["svix-signature"] as string
      }) as { type: string; data: any };

      // Handle event
      switch (event.type) {
        case "customer.product.created":
          if (event.data.scenario === "subscription_started") {
            await provisionAccess(event.data);
          }
          break;

        case "customer.product.updated":
          await updateFeatureAccess(event.data);
          break;

        case "customer.product.deleted":
          await revokeAccess(event.data);
          break;

        case "customer.threshold_reached":
          await sendUsageAlert(event.data);
          break;

        default:
          console.log("Unhandled event type:", event.type);
      }

      res.json({ received: true });
    } catch (error) {
      console.error("Webhook error:", error);
      res.status(400).json({ error: "Webhook processing failed" });
    }
  }
);
```
